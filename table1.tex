\begin{equation}
Y_{i,s,c} = \beta_0 + \beta_1 \theta_i + \beta_2 \Delta_{i} + \beta_3 \mathds{1}_{s,i} + \beta_4 \mathds{1}_{c,i} + \beta_5 \rho_i + \varepsilon_{i,s,c}
\end{equation}
\text{where:}
\begin{itemize}
  \item $Y_{i,s,c}$ is the referral low-SEB share for individual $i$, with SEB status $s$, in condition $c$
  \item $\theta_i$ is the standardized share of low-SEB in the classroom for individual $i$
  \item $\Delta_{i}$ is the standardized ratio of above median low-SEB share to the  low-SEB share at the classroom level
  \item $\mathds{1}_{s,i}$ indicates if individual $i$ is low-SEB
  \item $\mathds{1}_{c,i}$ indicates if individual $i$ is in the Quota condition
  \item $\rho_i$ is the standardized ratio of accurate low-SEB guesses to the actual low-SEB share in the \\ classroom for individual $i$
  \item $\varepsilon_{i,s,c}$ is the error term capturing all other factors influencing referral decisions
\end{itemize}
